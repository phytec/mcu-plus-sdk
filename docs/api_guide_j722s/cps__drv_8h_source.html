183 static inline uint32_t CPS_FldWrite(uint32_t mask, uint32_t shift, uint32_t reg_value, uint32_t value)
184 {
185  /* parasoft-begin-suppress MISRA2012-RULE-12_2-2 "shift ranges" */
186  uint32_t new_value = (value << shift) & mask;
187  /* parasoft-end-suppress MISRA2012-RULE-12_2-2 */
188 
189  new_value = (reg_value & ~mask) | new_value;
190  return (new_value);
