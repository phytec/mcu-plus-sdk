421 static inline void Dss_rmInfoInit(Dss_RmInfo *rmInfo)

431  {

435  {
436  rmInfo->isOverlayAvailable[i] = TRUE;

438  for(i=CSL_DSS_VP_ID_1; i<CSL_DSS_VP_ID_MAX; i++)
439  {
