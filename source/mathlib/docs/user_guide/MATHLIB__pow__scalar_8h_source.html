68 template <> inline float MATHLIB_log_pow_scalar_ci<float>(float a)
69 {
70  double ln2 = 0.693147180559945;
71  float MAXe = 3.402823466E+38f;
72  float c1 = -0.2302894f;
73  float c2 = 0.1908169f;
74  float c3 = -0.2505905f;
75  float c4 = 0.3333164f;
76  float c5 = -0.5000002f;
77  float pol, r1, r2, r3, r4, res;
78  double dr, frcpax, rcp, T;
79  int N, T_index;
80 
81  /* r = x * frcpa(x) -1 */
82  rcp = _rcpdp((double) a);
83  frcpax = _itod(_clr(_hi(rcp), 0u, 16u), 0u);
84  dr = (frcpax * (double) a) - 1.0;
85 
86  /* Polynomial p(r) that approximates ln(1+r) - r */
87  r1 = (float) dr;
88  r2 = r1 * r1;
89  r3 = r1 * r2;
90  r4 = r2 * r2;
91 
92  pol = (c5 * r2) + (c4 * r3) + (((c2 * r1) + c3 + (c1 * r2)) * r4);
93 
94  /* Reconstruction: result = T + r + p(r) */
95  N = (int) _extu(_hi(frcpax), 1u, 21u) - 1023;
96  T_index = (int) _extu(_hi(frcpax), 12u, 29u);
97  T = MATHLIB_logTable[T_index] - (ln2 * (double) N);
98  res = (float) (dr + T + (double) pol);
99 
100  if (a > MAXe) {
101  res = 88.72283905313f;
102  }
103 
104  return (res);
105 }

109 template <> inline double MATHLIB_log_pow_scalar_ci<double>(double a)
110 {
111  double Half = 0.5;
112  double MAXe = 1.7976931348623157e+308;
113  double srHalf = 0.70710678118654752440; /* sqrt(0.5) */
114  double a0 = -0.64124943423745581147e+2;
115  double a1 = 0.16383943563021534222e+2;
116  double a2 = -0.78956112887491257267e+0;
117  double b0 = -0.76949932108494879777e+3;
118  double b1 = 0.31203222091924532844e+3;
119  double b2 = -0.35667977739034646171e+2; /* Note b3 = 1.0 */
120  double c1 = 0.693359375; /* 355/512 */
121  double c2 = -2.121944400546905827679e-4;
122  double W, X, Y, Z;
123  double zn, zd;
124  double Rz, Sa, Bd, Cn, Da;
125  int N, exp_;
126  unsigned int upper;
127 
128  /* get unbiased exponent */
129  Y = a;
130  exp_ = (int) _extu(_hi(Y), 1u, 21u);
131  N = exp_ - 1022;
132 
133  /* force DP exp = 1022 if not zero */
134  upper = _clr(_hi(Y), 20u, 31u);
135  upper = 0x3fe00000u | upper;
136  Z = _itod(upper, _lo(Y));
137 
138  if (exp_ == 0) {
139  Z = 0.0;
140  }
141 
142  if (Z > srHalf) {
143  zn = (Z - Half) - Half;
144  zd = (Z * Half) + Half;
145  }
146  else {
147  zn = Z - Half;
148  zd = (zn * Half) + Half;
149  N = N - 1;
150  }
151 
152  X = divdpMod_powdpi(zn, zd);
153  W = X * X;
154  Bd = ((((W + b2) * W) + b1) * W) + b0;
155  Cn = (((W * a2) + a1) * W) + a0;
156  Rz = W * divdpMod_powdpi(Cn, Bd);
157  Sa = X + (X * Rz);
158  Cn = (double) N;
159  Da = ((Cn * c2) + Sa) + (Cn * c1);
160 
161  if (Y > MAXe) {
162  Da = 709.782712893384;
163  }
164 
165  return (Da);
166 }

172 template <> inline float MATHLIB_exp_pow_scalar_ci<float>(float a)
173 {
174  float log2_base_x16 = 1.442695041f * 16.0f;
175  float Halfe = 0.5f;
176  float LnMine = -87.33654475f;
177  float LnMaxe = 88.72283905f;
178  float Maxe = 3.402823466E+38f;
179  float c0 = 0.1667361910f;
180  float c1 = 0.4999999651f;
181  float c2 = 0.9999998881f;
182  float P1 = 0.04331970214844f;
183  float P2 = 1.99663646e-6f;
184  float pol, r, r2, r3, res;
185  unsigned int Ttemp;
186  unsigned int J, K;
187  float Nf;
188  int N;
189  double dT;
190 
191  /* Get N such that |N - x*16/ln(2)| is minimized */
192  Nf = (a * log2_base_x16) + Halfe;
193  N = (int) Nf; /* Cast from intermediate variable to appease MISRA */
194 
195  if ((a * log2_base_x16) < -Halfe) {
196  N--;
197  }
198 
199  /* Argument reduction, r, and polynomial approximation pol(r) */
200  r = (a - (P1 * (float) N)) - (P2 * (float) N);
201  r2 = r * r;
202  r3 = r * r2;
203 
204  pol = (r * c2) + ((r3 * c0) + (r2 * c1));
205 
206  /* Get index for ktable and jtable */
207  K = _extu((unsigned int ) N, 28u, 30u);
208  J = (unsigned int ) N & 0x3u;
209  dT = MATHLIB_kTable[K] * MATHLIB_jTable[J];
210 
211  /* Scale exponent to adjust for 2^M */
212  Ttemp = _hi(dT) + (((unsigned int) N >> 4) << 20);
213  dT = _itod(Ttemp, _lo(dT));
214 
215  res = (float) (dT * (1.0 + (double) pol));
216 
217  /* Early exit for small a */
218  if (_extu(_ftoi(a), 1u, 24u) < 114u) {
219  res = 1.0f + a;
220  }
221 
222  /* < LnMin returns 0 */
223  if (a < LnMine) {
224  res = 0.0f;
225  }
226 
227  /* > LnMax returns MAX */
228  if (a > LnMaxe) {
229  res = Maxe;
230  }
231 
232  return (res);
233 }

235 template <> inline double MATHLIB_exp_pow_scalar_ci<double>(double a)
236 {
237  double Halfe = 0.5;
238  double Maxe = 1.7976931348623157e+308;
239  double LnMaxe = 709.78271289338;
240  double LnMine = -708.3964185322641;
241  double a0e = 0.249999999999999993;
242  double a1e = 0.694360001511792852e-2;
243  double a2e = 0.165203300268279130e-4;
244  double b0e = 0.5;
245  double b1e = 0.555538666969001188e-1;
246  double b2e = 0.495862884905441294e-3;
247  double c1e = 0.693359375; /* 355/512 */
248  double C2e = -2.1219444005469058277e-4;
249  double L2e = 1.4426950408889634074; /* log (base 2) of e */
250  double Ye, Xe, We, Re, Se, Be, Ce, De;
251  int Ne, upper;
252 
253  Ye = a;
254  Ce = Ye * L2e;
255  Ne = _dpint(Ce);
256  Se = (double) Ne; /* double(int N) */
257  Xe = (Ye - (Se * c1e)) - (Se * C2e); /* range reduction */
258  We = Xe * Xe;
259  Be = (((b2e * We) + b1e) * We) + b0e; /* denominator */
260  De = ((((a2e * We) + a1e) * We) + a0e) * Xe; /* numerator */
261  Re = Halfe + divdpMod_powdpi(De, Be - De);
262  upper = 1024 + Ne;
263  upper = (int) _extu((unsigned int) upper, 20u, 0u);
264  Se = _itod((unsigned int) upper, 0x00000000u);
265  Ce = Re * Se; /* scale by power of 2 */
266  /* > LnMax returns MAX */
267  if (a > LnMaxe) {
268  Ce = Maxe;
269  }
270 
271  /* > LnMin returns 0 */
272  if (a < LnMine) {
273  Ce = 0.0;
274  }
275 
276  return (Ce);
277 }

283 template <> inline float MATHLIB_pow_scalar_ci<float>(float a, float b)
284 {
285  float arg, W, X2;
286  int y, Sign = 1;
287 
290  y = _spint(b);
291  if ((a < 0.0) && (b == (float) y)) {
292  if ((y % 2 ) != 0) {
293  Sign = -1;
294  }
295  }
296 
297  arg = _fabsf(a);
298  W = b * MATHLIB_log_pow_scalar_ci(arg);
299 
300  if (arg == 1.0f) {
301  W = 0.0f;
302  }
303 
304  X2 = (float ) Sign * MATHLIB_exp_pow_scalar_ci(W);
305 
306  if ((a < 0.0f) && (b != (float) y)) {
307  X2 = _itof(0x7fffffffu);
308  }
309 
310  if (a == 0.0f) {
311  X2 = (b >= 0.0f) ? 0.0f : _itof(0x7F800000u);
312  }
313 
314  if (b == 0.0f) {
315  X2 = 1.0f;
316  }
317 
318  return (X2);
319 }
