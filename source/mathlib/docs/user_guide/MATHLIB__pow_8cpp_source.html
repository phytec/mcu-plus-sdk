76 static inline c7x::double_vec cmn_DIVDP_opt(c7x::double_vec a, c7x::double_vec b)
77 {
78 
79  c7x::double_vec Two = (c7x::double_vec)(2.0f);
80  c7x::double_vec X;
81 
82  X = __recip(b);
83  X = X * (Two - (b * X));
84  X = X * (Two - (b * X));
85  X = X * (Two - (b * X));
86  X = a * X;
87 
88  return X;
89 }

99  size_t numBlocks = 0; // compute loop's iteration count
100  size_t remNumBlocks = 0; // when numBlocks is not a multiple of SIMD width
101

104

107 
108  MATHLIB_SE0SA01DSequentialInit(&se0Params, &sa0Params, length, pDst, pDst);
109 
110  // calculate compute loop's iteration counter
111  numBlocks = length / c7x::element_count_of<vec>::value;
112  remNumBlocks = length % c7x::element_count_of<vec>::value;
113  if (remNumBlocks) {
114  numBlocks++;
115  }
116 
117  // open SE0, SE1, and SA0 for reading and writing operands
118  MATHLIB_SE0SA0Open(&se0Params, &sa0Params, pSrc0);
119 
120  vec Half, MAXe, srHalf, Half_sq, MINe, a0, a1, a2, b0, b1, b2, c1_vec, c2_vec, W, X, Y, Z, zn, zd, Rz, Sa, Bd, Cn, Da;
121 
122  Half = (vec) 0.5;
123  Half_sq = (vec) 0.5 * 0.5;
124  MAXe = (vec) 1.7976931348623157e+308;
125  srHalf = (vec) 0.70710678118654752440; /* sqrt(0.5) */
126  MINe = (vec) 2.2250738585072014e-308;
127  a0 = (vec) -0.64124943423745581147e+2;
128  a1 = (vec) 0.16383943563021534222e+2;
129  a2 = (vec) -0.78956112887491257267e+0;
130  b0 = (vec) -0.76949932108494879777e+3;
131  b1 = (vec) 0.31203222091924532844e+3;
132  b2 = (vec) -0.35667977739034646171e+2; /* Note b3 = 1.0 */
133  c1_vec = (vec) 0.693359375; /* 355/512 */
134  c2_vec = (vec) -2.121944400546905827679e-4;
135 
136  c7x::long_vec long_zero_vec = (c7x::long_vec) 0;
137  vec double_zero_vec = (vec) 0.0;
138  vec outMAX = (vec)(308.254715974092);
139 
140  for (size_t i = 0; i < numBlocks; i++) {
141 
142  vec a = c7x::strm_eng<0, vec>::get_adv();
143  Y = __abs(a);
144  c7x::long_vec exp_ = c7x::as_long_vec((c7x::as_ulong_vec(Y) << 1) >> 53);
145 
146  c7x::ulong_vec upper = c7x::as_ulong_vec(Y) & (0x000FFFFF00000000u);
147  upper = 0x3FE0000000000000u | upper;
148 
149  Z = c7x::as_double_vec((0x00000000FFFFFFFFu & c7x::as_ulong_vec(Y)) | upper);
150 
151  __vpred cmp1 = __cmp_eq_pred(exp_, long_zero_vec);
152  Z = __select(cmp1, double_zero_vec, Z);
153 
154  vec z_minus_half = Z - Half;
155  vec z_mul_half = (Z * Half) + Half;
156  __vpred cmp2 = __cmp_lt_pred(srHalf, Z);
157  zn = __select(cmp2, (z_minus_half - Half), z_minus_half);
158  zd = __select(cmp2, z_mul_half, (z_mul_half - Half_sq));
159 
160  X = cmn_DIVDP_opt(zn, zd);
161  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
162  vec * addr = c7x::strm_agen<0, vec>::get_adv(pDst);
163  __vstore_pred(tmp, addr, X);
164  }
165  MATHLIB_SE0SA0Close();
166 
167  MATHLIB_SE0SA0Open(&se0Params, &sa0Params, pDst);
168 
169  for (size_t i = 0; i < numBlocks; i++) {
170 
171  X = c7x::strm_eng<0, vec>::get_adv();
172 
173  W = X * X;
174  Bd = ((((W + b2) * W) + b1) * W) + b0;
175  Cn = (((W * a2) + a1) * W) + a0;
176  Rz = W * cmn_DIVDP_opt(Cn, Bd);
177  Sa = X + (X * Rz);
178  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
179  vec * addr = c7x::strm_agen<0, vec>::get_adv(pDst);
180  __vstore_pred(tmp, addr, Sa);
181  }
182 
183  MATHLIB_SE0SA0Close();
184 
185  MATHLIB_SE0SA0Open(&se0Params, &sa0Params, pSrc0);
186 
187  __SE1_OPEN(pDst, se0Params);
188  for (size_t i = 0; i < numBlocks; i++) {
189 
190  vec a = c7x::strm_eng<0, vec>::get_adv();
191  Sa = c7x::strm_eng<1, vec>::get_adv();
192 
193  Y = a;
194  c7x::long_vec exp_ = c7x::as_long_vec((c7x::as_ulong_vec(Y) << 1) >> 53);
195  c7x::long_vec N_vec = exp_ - 1022;
196 
197  c7x::ulong_vec upper = c7x::as_ulong_vec(Y) & (0x000FFFFF00000000u);
198  upper = 0x3FE0000000000000u | upper;
199 
200  Z = c7x::as_double_vec((0x00000000FFFFFFFFu & c7x::as_ulong_vec(Y)) | upper);
201 
202  __vpred cmp1 = __cmp_eq_pred(exp_, long_zero_vec);
203  Z = __select(cmp1, double_zero_vec, Z);
204 
205  __vpred cmp2 = __cmp_lt_pred(srHalf, Z);
206 
207  N_vec = __select(cmp2, N_vec, (N_vec - 1));
208 
209  Cn = __low_int_to_double(c7x::as_int_vec(N_vec));
210  Da = ((Cn * c2_vec) + Sa) + (Cn * c1_vec);
211 
212  /**********************************************************************/
213  /* Bounds checking */
214  /**********************************************************************/
215 
216  // if (Y < MINe) {
217  // Da = -MAXe;
218  // }
219  __vpred cmp_min_vec = __cmp_lt_pred(Y, MINe);
220  Da = __select(cmp_min_vec, -MAXe, Da);
221 
222  // if (Y > MAXe) {
223  // Da = 308.254715974092;
224  // }
225  __vpred cmp_max_vec = __cmp_lt_pred(MAXe, Y);
226  Da = __select(cmp_max_vec, outMAX, Da);
227 
228  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
229  vec * addr = c7x::strm_agen<0, vec>::get_adv(pDst);
230  __vstore_pred(tmp, addr, Da);
231  }
232  __SE1_CLOSE();
233  MATHLIB_SE0SA0Close();
234 }

290  __SE1_OPEN(pSrc0, se1Params);
291  __SA0_OPEN(sa0Params);
292 
293  vec zeroVec, one, X2;
294 
295  zeroVec = (vec) 0.0f;
296  one = (vec) 1.0f;
297  c7x::long_vec zero_int = (c7x::long_vec)(0);
298  c7x::long_vec one_int = (c7x::long_vec)(1);
299  c7x::int_vec Sign = (c7x::int_vec)(1);
300  c7x::int_vec negate_Sign = (c7x::int_vec)(-1);

302  c7x::long_vec valueL1 = (c7x::long_vec)(0X7fffffffffffffff);
303  c7x::long_vec valueL2 = (c7x::long_vec)(0X7ff0000000000000);
304 
305  for (size_t i = 0; i < numBlocks; i++) {
306  vec inVec0 = c7x::strm_eng<1, vec>::get_adv();
307  vec inVec1 = c7x::strm_eng<1, vec>::get_adv();
308  vec W = c7x::strm_eng<0, vec>::get_adv();
309 
310  c7x::int_vec y = __double_to_int(inVec1);
311 
312  __vpred cmp_cond1 = __cmp_lt_pred(inVec0, zeroVec);
313  __vpred cmp_cond2 = __cmp_eq_pred(inVec1, __low_int_to_double(y));
314 
315  cmp_cond2 = __and(cmp_cond2, cmp_cond1);
316 
317  c7x::long_vec y_long = c7x::as_long_vec(y);
318  __vpred cmp_cond3 = __cmp_eq_pred((y_long & one_int), zero_int);
319

321 
322  Sign = __select(cmp_cond3, negate_Sign, Sign);
323 
324  X2 = __low_int_to_double(Sign) * W;
325 
326  __vpred cmp_cond5 = __cmp_eq_pred(inVec1, __low_int_to_double(y));
327  cmp_cond5 = __and(cmp_cond1, __negate(cmp_cond5));
328  X2 = __select(cmp_cond5, c7x::as_double_vec(valueL1), X2);
329 
330  __vpred cmp_cond6 = __cmp_eq_pred(inVec0, zeroVec);
331  __vpred cmp_cond7 = __cmp_lt_pred(inVec1, zeroVec);
332  cmp_cond7 = __and(cmp_cond6, cmp_cond7);
333  X2 = __select(cmp_cond7, c7x::as_double_vec(valueL2), X2);
334 
335  // Store log computation result in
336  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
337  vec * addr = c7x::strm_agen<0, vec>::get_adv(pDst);
338  __vstore_pred(tmp, addr, X2);
339  }
340  MATHLIB_SE0SE1SA0Close();
341  return status;

346  __SA_TEMPLATE_v1 *restrict sa0Params,

359

361  numBlocks = length / c7x::element_count_of<vec>::value;
362  remNumBlocks = length % c7x::element_count_of<vec>::value;
363  if (remNumBlocks) {
364  numBlocks++;
365  }
366 
367  // open SE0, SE1, and SA0 for reading and writing operands
368  MATHLIB_SE0SA0Open(se0Params, sa0Params, pDst);
369 
370  __SE1_OPEN(pSrc1, *se0Params);
371 
372  vec Halfe, Maxe, LnMaxe, LnMine, a0e, a1e, a2e, b0e, b1e, b2e, c1e, C2e, L2e, Zero;
373 
374  Halfe = (vec) 0.5;
375  Maxe = (vec) 1.7976931348623157e+308;
376  LnMaxe = (vec) 709.78271289338;
377  LnMine = (vec) -708.3964185322641;
378  a0e = (vec) 0.249999999999999993;
379  a1e = (vec) 0.694360001511792852e-2;
380  a2e = (vec) 0.165203300268279130e-4;
381  b0e = (vec) 0.5;
382  b1e = (vec) 0.555538666969001188e-1;
383  b2e = (vec) 0.495862884905441294e-3;
384  c1e = (vec) 0.693359375; /* 355/512 */
385  C2e = (vec) -2.1219444005469058277e-4;
386  L2e = (vec) 1.4426950408889634074;
387  Zero = (vec) 0.0f;
388 
389  vec Ye, Xe, We, Re, Se, Be, Ce, De;
390 
391  // compute loop to perform vector pow
392  for (size_t i = 0; i < numBlocks; i++) {
393  vec invec = c7x::strm_eng<0, vec>::get_adv();
394  vec b = c7x::strm_eng<1, vec>::get_adv();
395 
396  Ye = invec * b;
397  Ce = Ye * L2e;
398 
399  c7x::int_vec Ne = __double_to_int(Ce);
400 
401  Se = __low_int_to_double(Ne);
402  Xe = (Ye - (Se * c1e)) - (Se * C2e); /* range reduction */
403 
404  We = Xe * Xe;
405  Be = (((b2e * We) + b1e) * We) + b0e; /* denominator */
406  De = ((((a2e * We) + a1e) * We) + a0e) * Xe; /* numerator */
407  Re = Halfe + cmn_DIVDP_opt(De, Be - De);
408 
409  Ye = invec;
410 
411  Ce = Ye * L2e;
412 
413  c7x::int_vec upper = 1024 + (Ne);
414  upper = c7x::as_int_vec((c7x::as_uint_vec(upper) << 20));
415  Se = c7x::as_double_vec(c7x::as_long_vec(upper) << 32);
416 
417  Ce = Re * Se;
418 
419  __vpred cmp_Max = __cmp_lt_pred(LnMaxe, invec);
420  Ce = __select(__negate(cmp_Max), Ce, Maxe);
421 
422  __vpred cmp_Min = __cmp_lt_pred(invec, LnMine);
423  Ce = __select(cmp_Min, Zero, Ce);
424 
425  // Store exp computation result in
426  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
427  vec * addr = c7x::strm_agen<0, vec>::get_adv(pDst);
428  __vstore_pred(tmp, addr, Ce);
429  }
430 
431  MATHLIB_SE0SA0Close();
432  __SE1_CLOSE();
433  return status;
