968 static inline bool CpswProxy_isSwitchPort(EthRemoteCfg_VirtPort portId)
969 {
970  return ((portId >= ETHREMOTECFG_SWITCH_PORT_0) && (portId <= ETHREMOTECFG_SWITCH_PORT_LAST));
971 }

983 }

995 static inline uint32_t CpswProxy_getPortNum(EthRemoteCfg_VirtPort portId)
996 {
997  uint32_t portNum;
998 
999  if (CpswProxy_isSwitchPort(portId))
1000  {
1001  portNum = (uint32_t)portId;
1002  }
1003  else
1004  {
1005  portNum = (uint32_t)(portId - ETHREMOTECFG_MAC_PORT_1) + 1U;
1006  }
1007 
1008  return portNum;
1009 }
