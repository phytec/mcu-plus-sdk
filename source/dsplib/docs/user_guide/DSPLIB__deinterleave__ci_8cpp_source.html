166 template <typename dataType>
167 inline void deinterleave_init(DSPLIB_kernelHandle handle,
168  const DSPLIB_bufParams2D_t *bufParamsIn,
169  const DSPLIB_bufParams2D_t *bufParamsOut)
170 {
171  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Entering template function to init SE for 32-bit data");
172 
173  DSPLIB_deinterleave_PrivArgs *pKerPrivArgs = (DSPLIB_deinterleave_PrivArgs *) handle;
174  uint8_t *pBlock = pKerPrivArgs->bufPblock;
175  uint32_t widthIn = pKerPrivArgs->widthIn;
176  uint32_t heightIn = pKerPrivArgs->heightIn;
177  int32_t strideOut = bufParamsOut->stride_y;
178  int32_t strideIn = bufParamsIn->stride_y;
179  int32_t dataSize = sizeof(dataType);
180  __SE_TEMPLATE_v1 se0Params = __gen_SE_TEMPLATE_v1();
181  __SA_TEMPLATE_v1 sa0Params = __gen_SA_TEMPLATE_v1();
182  typedef typename c7x::make_full_vector<dataType>::type vec;
183  int32_t eleCount = c7x::element_count_of<vec>::value;
184  __SE_ELETYPE SE_ELETYPE = c7x::se_eletype<vec>::value;
185  __SE_VECLEN SE_VECLEN = c7x::se_veclen<vec>::value;
186  __SA_VECLEN SA_VECLEN = c7x::sa_veclen<vec>::value;
187 
188  if (widthIn == 1) {
189 
190  if ((bufParamsIn->data_type == DSPLIB_FLOAT64) || ((bufParamsIn->data_type == DSPLIB_INT64)) ||
191  ((bufParamsIn->data_type == DSPLIB_UINT64))) {
192  int32_t iter = (heightIn + (eleCount * 2) - 1) / (eleCount * 2);
193 
194  se0Params.ICNT0 = widthIn;
195  se0Params.ICNT1 = (heightIn > (uint32_t) eleCount) ? eleCount : heightIn;
196  se0Params.DIM1 = strideIn / dataSize;
197  se0Params.ICNT2 = DSPLIB_ceilingDiv(heightIn, eleCount);
198  se0Params.DIM2 = (strideIn / dataSize) * eleCount;
199  se0Params.DIMFMT = __SE_DIMFMT_3D;
200  se0Params.TRANSPOSE = __SE_TRANSPOSE_64BIT;
201  se0Params.ELETYPE = SE_ELETYPE;
202  se0Params.VECLEN = SE_VECLEN;
203 
204  sa0Params.ICNT0 = heightIn;
205  sa0Params.VECLEN = SA_VECLEN;
206  sa0Params.DIMFMT = __SA_DIMFMT_1D;
207  pKerPrivArgs->loopCount = DSPLIB_ceilingDiv(heightIn, eleCount);
208 
209  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
210  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
211  *(int32_t *) ((uint8_t *) pBlock + (2 * SE_SA0_PARAM_OFFSET)) = iter;
212  }
213  else if ((bufParamsIn->data_type == DSPLIB_FLOAT32) || ((bufParamsIn->data_type == DSPLIB_INT32)) ||
214  ((bufParamsIn->data_type == DSPLIB_UINT32))) {
215  int32_t iter = (heightIn + (eleCount * 2) - 1) / (eleCount * 2);
216  se0Params.TRANSPOSE = __SE_TRANSPOSE_32BIT;
217  se0Params.ICNT0 = widthIn;
218  se0Params.ICNT1 = (heightIn > (uint32_t) eleCount) ? eleCount : heightIn;
219  se0Params.DIM1 = strideIn / dataSize;
220  se0Params.ICNT2 = DSPLIB_ceilingDiv(heightIn, eleCount);
221  se0Params.DIM2 = (strideIn / dataSize) * eleCount;
222  se0Params.DIMFMT = __SE_DIMFMT_3D;
223  se0Params.ELETYPE = SE_ELETYPE;
224  se0Params.VECLEN = SE_VECLEN;
225 
226  sa0Params.ICNT0 = heightIn;
227  sa0Params.VECLEN = SA_VECLEN;
228  sa0Params.DIMFMT = __SA_DIMFMT_1D;
229  pKerPrivArgs->loopCount = DSPLIB_ceilingDiv(heightIn, eleCount);
230 
231  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
232  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
233  *(int32_t *) ((uint8_t *) pBlock + (2 * SE_SA0_PARAM_OFFSET)) = iter;
234  }
235  else if (((bufParamsIn->data_type == DSPLIB_INT16)) || ((bufParamsIn->data_type == DSPLIB_UINT16))) {
236  se0Params.TRANSPOSE = __SE_TRANSPOSE_32BIT;
237  se0Params.ICNT0 = widthIn;
238  se0Params.ICNT1 = (heightIn > (uint32_t) (eleCount / 2)) ? (eleCount / 2) : heightIn;
239  se0Params.ICNT2 = DSPLIB_ceilingDiv(heightIn, eleCount);
240  se0Params.DIM2 = (strideIn / dataSize) * eleCount;
241  se0Params.ELETYPE = SE_ELETYPE;
242  se0Params.VECLEN = SE_VECLEN;
243  se0Params.DIM1 = strideIn / dataSize;
244  se0Params.DIMFMT = __SE_DIMFMT_3D;
245 
246  sa0Params.ICNT0 = heightIn;
247  sa0Params.DIMFMT = __SA_DIMFMT_1D;
248  sa0Params.VECLEN = SA_VECLEN;
249 
250  pKerPrivArgs->loopCount = DSPLIB_ceilingDiv(heightIn, eleCount);
251 
252  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
253  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
254  }
255  else if ((bufParamsIn->data_type == DSPLIB_INT8) || (bufParamsIn->data_type == DSPLIB_UINT8)) {
256  uint32_t outEleCount = eleCount / 4;
257 
258  se0Params.TRANSPOSE = __SE_TRANSPOSE_32BIT;
259  se0Params.ELETYPE = SE_ELETYPE;
260  se0Params.VECLEN = SE_VECLEN;
261  se0Params.ICNT0 = widthIn;
262  se0Params.ICNT1 = outEleCount;
263  se0Params.DIM1 = strideIn;
264  se0Params.DIMFMT = __SE_DIMFMT_3D;
265  se0Params.ICNT2 = DSPLIB_ceilingDiv(heightIn, outEleCount);
266  se0Params.DIM2 = strideIn * outEleCount;
267 
268  sa0Params.VECLEN = SA_VECLEN;
269  sa0Params.ICNT0 = outEleCount;
270  sa0Params.ICNT1 = DSPLIB_ceilingDiv(heightIn, outEleCount);
271  sa0Params.DIM1 = outEleCount;
272  sa0Params.DIMFMT = __SA_DIMFMT_2D;
273 
274  uint8_t mask[64] = {
275  0, 4, 8, 12, 16, 20, 24, 28, // 0
276  2, 9, 10, 11, 3, 13, 14, 15, // 1
277  1, 17, 18, 19, 5, 21, 22, 23, // 2
278  6, 25, 26, 27, 7, 29, 30, 31, // 3
279  32, 33, 34, 35, 36, 37, 38, 39, // 4
280  40, 41, 42, 43, 44, 45, 46, 47, // 5
281  48, 49, 50, 51, 52, 53, 54, 55, // 6
282  56, 57, 58, 59, 60, 61, 62, 63 // 7
283  };
284  pKerPrivArgs->loopCount = DSPLIB_ceilingDiv(heightIn, (eleCount / 4));
285 
286  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
287  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
288  memcpy((uint8_t *) pBlock + MASK_OFFFSET, mask, 64);
289  }
290  else {
291  se0Params.ICNT0 = 1;
292  se0Params.DIM1 = 1;
293  se0Params.ICNT1 = heightIn * widthIn;
294  se0Params.DIMFMT = __SE_DIMFMT_2D;
295  se0Params.ELETYPE = SE_ELETYPE;
296  se0Params.VECLEN = SE_VECLEN;
297 
298  sa0Params.ICNT0 = 1;
299  sa0Params.ICNT1 = heightIn * widthIn;
300  sa0Params.DIM1 = 1;
301 
302  sa0Params.VECLEN = SA_VECLEN;
303  sa0Params.DIMFMT = __SA_DIMFMT_2D;
304 
305  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
306  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
307  }
308  }
309  else {
310  se0Params.ICNT0 = widthIn * heightIn;
311  se0Params.DIMFMT = __SE_DIMFMT_1D;
312  se0Params.ELETYPE = SE_ELETYPE;
313  se0Params.VECLEN = SE_VECLEN;
314  se0Params.DECIM = __SE_DECIM_2;
315 
316  sa0Params.ICNT0 = eleCount;
317  sa0Params.ICNT1 = widthIn;
318  sa0Params.ICNT2 = DSPLIB_ceilingDiv(heightIn, eleCount);
319  sa0Params.DIM1 = strideOut / dataSize;
320  sa0Params.DIM2 = eleCount;
321  sa0Params.VECLEN = SA_VECLEN;
322  sa0Params.DIMFMT = __SA_DIMFMT_3D;
323  if (bufParamsIn->data_type == DSPLIB_UINT8 || bufParamsIn->data_type == DSPLIB_INT8) {
324 
325  pKerPrivArgs->loopCount = DSPLIB_ceilingDiv(DSPLIB_ceilingDiv(widthIn * heightIn, eleCount), 2);
326  }
327  else {
328  pKerPrivArgs->loopCount = DSPLIB_ceilingDiv(DSPLIB_ceilingDiv(widthIn * heightIn, eleCount), 2);
329  }
330 
331  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
332  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
333  }
334 
335  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Exiting template function to init SE ");
336 }

397 template <typename dataType>
398 inline void deinterleave_compute_64_32bit(DSPLIB_kernelHandle handle, void *restrict pIn, void *restrict pOut)
399 {
400  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Entering template function for Transpose 64/32-bit data");
401 
402  DSPLIB_deinterleave_PrivArgs *pKerPrivArgs = (DSPLIB_deinterleave_PrivArgs *) handle;
403  uint32_t widthIn = pKerPrivArgs->widthIn;
404  __SE_TEMPLATE_v1 se0Params;
405  __SA_TEMPLATE_v1 sa0Params;
406  uint8_t *pBlock = pKerPrivArgs->bufPblock;
407  se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET);
408  sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET);
409  dataType *restrict pInLocal = (dataType *) pIn;
410  dataType *restrict pOutLocal = (dataType *) pOut;
411  typedef typename c7x::make_full_vector<dataType>::type vec;
412  int32_t loopCount = pKerPrivArgs->loopCount;
413  typedef typename c7x::make_full_vector<dataType>::type vec;
414 
415  using vecType = typename c7x_vec_type<dataType>::type;
416 
417  DSPLIB_DEBUGPRINTFN(0, "pInLocal: %p pOutLocal: %p loopCount: %d\n", pInLocal, pOutLocal, loopCount);
418  __SE0_OPEN(pInLocal, se0Params);
419  __SA0_OPEN(sa0Params);
420 
421  if (widthIn == 1) {
422 
423  for (int32_t i = 0; i < loopCount; i++) {
424  vec loadVec1 = c7x::strm_eng<0, vec>::get_adv();
425 
426  __vpred predTemp = c7x::strm_agen<0, vec>::get_vpred();
427  vec *pStoreVec = c7x::strm_agen<0, vec>::get_adv(pOutLocal);
428  __vstore_pred(predTemp, pStoreVec, loadVec1);
429  }
430  __SE0_CLOSE();
431  __SA0_CLOSE();
432  }
433  else {
434  __SE1_OPEN(pInLocal + 1, se0Params);
435 
436  for (int32_t i = 0; i < loopCount; i++) {
437  vec loadVec1 = c7x::strm_eng<0, vec>::get_adv();
438  vec loadVec2 = c7x::strm_eng<1, vec>::get_adv();
439 
440  loadVec1 = c7x_vec_cast<vecType>::convert(loadVec1);
441  loadVec2 = c7x_vec_cast<vecType>::convert(loadVec2);
442 
443  __vpred predTemp = c7x::strm_agen<0, vec>::get_vpred();
444  vec *pStoreVec = c7x::strm_agen<0, vec>::get_adv(pOutLocal);
445  __vstore_pred(predTemp, pStoreVec, loadVec1);
446 
447  __vpred predTemp1 = c7x::strm_agen<0, vec>::get_vpred();
448  vec *pStoreVec1 = c7x::strm_agen<0, vec>::get_adv(pOutLocal);
449  __vstore_pred(predTemp1, pStoreVec1, loadVec2);
450  }
451  }
452 
453  __SE0_CLOSE();
454  __SA0_CLOSE();
455 
456  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Exiting template function for Transpose 64/32-bit data");
457 }

461 {
462  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Entering template function for Transpose 16-bit data");

464  DSPLIB_deinterleave_PrivArgs *pKerPrivArgs = (DSPLIB_deinterleave_PrivArgs *) handle;

467  uint32_t dataSize = sizeof(dataType);
468  __SE_TEMPLATE_v1 se0Params;
469  __SA_TEMPLATE_v1 sa0Params;
470  uint8_t *pBlock = pKerPrivArgs->bufPblock;
471  se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET);
472  sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET);
473  dataType *restrict pInLocal = (dataType *) pIn;
474  dataType *restrict pOutLocal = (dataType *) pOut;

590 void DSPLIB_deinterleave_perfEst(DSPLIB_kernelHandle handle,
591  uint64_t *archCycles,
592  uint64_t *estCycles,
593  uint64_t dataType)
594 {
595  DSPLIB_deinterleave_PrivArgs *pKerPrivArgs = (DSPLIB_deinterleave_PrivArgs *) handle;
596 
597  uint64_t deinterleaveStartupCycles = 0;
598  uint64_t deinterleaveTeardownCycles = 0;
599  uint64_t deinterleaveOperationCycles = 0;
600  uint64_t deinterleaveOverheadCycles = 0;
601  uint32_t widthIn = pKerPrivArgs->widthIn;
602  uint32_t strideIn = pKerPrivArgs->strideIn;
603 
604 
605  if ((widthIn == 1u) || (widthIn == 2u && (uint32_t) widthIn == (uint32_t) strideIn / DSPLIB_sizeof(dataType))) {
606 
607  if ((dataType == DSPLIB_FLOAT32) || (dataType == DSPLIB_FLOAT64) || (dataType == DSPLIB_INT32) ||
608  (dataType == DSPLIB_UINT32) || (dataType == DSPLIB_INT64) || (dataType == DSPLIB_UINT64)) {
609  if (widthIn == 1) {
610  uint32_t loopCount = pKerPrivArgs->loopCount;
611  deinterleaveOverheadCycles = 25;
612  deinterleaveStartupCycles = 4;
613  deinterleaveOperationCycles = 1 + loopCount * 1;
614  deinterleaveTeardownCycles = 1;
615  }
616  else {
617  uint32_t loopCount = pKerPrivArgs->loopCount;
618  deinterleaveOverheadCycles = 25 + 16;
619  deinterleaveStartupCycles = 4;
620  deinterleaveOperationCycles = 1 + loopCount * 2;
621  deinterleaveTeardownCycles = 4;
622  }
623  }
624  if ((dataType == DSPLIB_INT16) || (dataType == DSPLIB_UINT16)) {
625  if (widthIn == 1) {
626  uint32_t loopCount = pKerPrivArgs->loopCount;
627  deinterleaveOverheadCycles = 16;
628  deinterleaveStartupCycles = 14;
629  deinterleaveOperationCycles = 4 + loopCount * 1;
630  deinterleaveTeardownCycles = 3;
631  }
632  else {
633  uint32_t loopCount = pKerPrivArgs->loopCount;
634  deinterleaveOverheadCycles = 16;
635  deinterleaveStartupCycles = 16;
636  deinterleaveOperationCycles = 1 + loopCount * 2;
637  deinterleaveTeardownCycles = 2;
638  }
639  }
640  else {
641  if (widthIn == 1) {
642  uint32_t loopCount = pKerPrivArgs->loopCount;
643  deinterleaveOverheadCycles = 23;
644  deinterleaveStartupCycles = 4;
645  deinterleaveOperationCycles = 4 + loopCount * 1;
646  deinterleaveTeardownCycles = 2;
647  }
648  else {
649  uint32_t loopCount = pKerPrivArgs->loopCount;
650  deinterleaveOverheadCycles = 23;
651  deinterleaveStartupCycles = 15;
652  deinterleaveOperationCycles = 1 + loopCount * 2;
653  deinterleaveTeardownCycles = 1;
654  }
655  }
656  deinterleaveOverheadCycles += deinterleaveStartupCycles + deinterleaveTeardownCycles;
657  *estCycles = deinterleaveOperationCycles + deinterleaveOverheadCycles;
658  *archCycles = deinterleaveOperationCycles;
659  }
660  else {
661  DSPLIB_matTrans_perfEst(handle,archCycles,estCycles,dataType,widthIn);
662  }
