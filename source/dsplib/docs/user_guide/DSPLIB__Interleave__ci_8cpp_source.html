378 template <typename dataType>
379 inline void interleave_compute(DSPLIB_kernelHandle handle, void *restrict pIn, void *restrict pOut)
380 {
381  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Entering template function for Transpose 64/32-bit data");
382 
383  DSPLIB_interleave_PrivArgs *pKerPrivArgs = (DSPLIB_interleave_PrivArgs *) handle;
384  uint32_t heightIn = pKerPrivArgs->heightIn;
385  int32_t strideIn = pKerPrivArgs->strideIn;
386  uint32_t dataSize = sizeof(dataType);
387  uint32_t widthIn = pKerPrivArgs->widthIn;
388  __SE_TEMPLATE_v1 se0Params;
389  __SA_TEMPLATE_v1 sa0Params;
390  uint8_t *pBlock = pKerPrivArgs->bufPblock;
391  se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET);
392  sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET);
393  dataType *restrict pInLocal = (dataType *) pIn;
394  dataType *restrict pOutLocal = (dataType *) pOut;
395  dataType *restrict pOutLocal1 = (dataType *) pOut;
396  typedef typename c7x::make_full_vector<dataType>::type vec;
397  int32_t loopCount = pKerPrivArgs->loopCount;
398  using vecType = typename c7x_vec_type<dataType>::type;
399  strideIn = strideIn / dataSize;
400  int32_t eleCount = c7x::element_count_of<vec>::value;
401 
402  DSPLIB_DEBUGPRINTFN(0, "pInLocal: %p pOutLocal: %p loopCount: %d\n", pInLocal, pOutLocal, loopCount);
403  __SE0_OPEN(pInLocal, se0Params);
404  __SA0_OPEN(sa0Params);
405  __SA1_OPEN(sa0Params);
406 
407  if (heightIn == 1) {
408  pOutLocal1 += 1;
409  loopCount = DSPLIB_ceilingDiv(widthIn, eleCount);
410  for (int32_t i = 0; i < loopCount; i++) {
411  vec vec1 = c7x::strm_eng<0, vec>::get_adv();
412 
413  for (int32_t i = 0; i < eleCount; i += 4) {
414 
415  dataType data = vec1.s[0 + i];
416 
417  dataType data3 = vec1.s[1 + i];
418 
419  __vpred dstWr = c7x::strm_agen<0, dataType>::get_vpred();
420 
421  dataType *dstWrPtr = c7x::strm_agen<0, dataType>::get_adv(pOutLocal);
422 
423  __vstore_pred(dstWr, dstWrPtr, data);
424 
425  __vpred dstWr2 = c7x::strm_agen<1, dataType>::get_vpred();
426 
427  dataType *dstWrPtr2 = c7x::strm_agen<1, dataType>::get_adv(pOutLocal1);
428 
429  __vstore_pred(dstWr2, dstWrPtr2, data3);
430 
431  data = vec1.s[2 + i];
432  data3 = vec1.s[3 + i];
433 
434  dstWr = c7x::strm_agen<0, dataType>::get_vpred();
435 
436  dstWrPtr = c7x::strm_agen<0, dataType>::get_adv(pOutLocal);
437 
438  __vstore_pred(dstWr, dstWrPtr, data);
439 
440  dstWr2 = c7x::strm_agen<1, dataType>::get_vpred();
441 
442  dstWrPtr2 = c7x::strm_agen<1, dataType>::get_adv(pOutLocal1);
443 
444  __vstore_pred(dstWr2, dstWrPtr2, data3);
445  }
446  }
447  }
448  else {
449 
450  for (int32_t i = 0; i < loopCount; i++) {
451  vec loadVec1 = c7x::strm_eng<0, vec>::get_adv();
452  vec loadVec2 = c7x::strm_eng<0, vec>::get_adv();
453 
454  loadVec1 = c7x_vec_cast<vecType>::convert(loadVec1);
455  loadVec2 = c7x_vec_cast<vecType>::convert(loadVec2);
456 
457  __vpred predTemp = c7x::strm_agen<0, vec>::get_vpred();
458  vec *pStoreVec = c7x::strm_agen<0, vec>::get_adv(pOutLocal);
459  __vstore_pred_interleave_low_low(predTemp, pStoreVec, (loadVec1), (loadVec2));
460 
461  __vpred predTemp1 = c7x::strm_agen<0, vec>::get_vpred();
462  vec *pStoreVec1 = c7x::strm_agen<0, vec>::get_adv(pOutLocal);
463  __vstore_pred_interleave_high_high(predTemp1, pStoreVec1, loadVec1, loadVec2);
464  }
465  }
466 
467  __SE0_CLOSE();
468  __SA0_CLOSE();
469 
470  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Exiting template function for interleave compute");
471 }

473 template <typename dataType>
474 inline void interleave_compute_fp(DSPLIB_kernelHandle handle, void *restrict pIn, void *restrict pOut)
475 {
476  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Entering template function for Transpose 16-bit data");
477 
478  DSPLIB_interleave_PrivArgs *pKerPrivArgs = (DSPLIB_interleave_PrivArgs *) handle;
479  uint32_t heightIn = pKerPrivArgs->heightIn;
480  __SE_TEMPLATE_v1 se0Params;
481  __SA_TEMPLATE_v1 sa0Params;
482  uint32_t widthIn = pKerPrivArgs->widthIn;
483 
484  uint8_t *pBlock = pKerPrivArgs->bufPblock;
485  se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET);
486  sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET);
487  dataType *restrict pInLocal = (dataType *) pIn;
488  dataType *restrict pOutLocal = (dataType *) pOut;
489  dataType *restrict pOutLocal1 = (dataType *) pOut;
490 
491  typedef typename c7x::make_full_vector<dataType>::type vec;
492  uint32_t loopCount = pKerPrivArgs->loopCount;
493  using vecType = typename c7x_vec_type<dataType>::type;
494  int32_t eleCount = c7x::element_count_of<vec>::value;
495 
496  DSPLIB_DEBUGPRINTFN(0, "pInLocal: %p pOutLocal: %p loopCount: %d\n", pInLocal, pOutLocal, loopCount);
497  __SE0_OPEN(pInLocal, se0Params);
498  __SA0_OPEN(sa0Params);
499  __SA1_OPEN(sa0Params);
500 
501  uint8_t *mask = (uint8_t *) pBlock + MASK_OFFSET;
502 
503  if (heightIn == 1) {
504 
505  loopCount = DSPLIB_ceilingDiv(widthIn, eleCount);
506  pOutLocal1 += 1;
507 
508  for (uint32_t i = 0; i < loopCount; i++) {
509  vec vec1 = c7x::strm_eng<0, vec>::get_adv();
510  for (int i = 0; i < eleCount; i += 4) {
511 
512  dataType data = vec1.s[0 + i];
513 
514  dataType data3 = vec1.s[1 + i];
515 
516  __vpred dstWr = c7x::strm_agen<0, dataType>::get_vpred();
517 
518  dataType *dstWrPtr = c7x::strm_agen<0, dataType>::get_adv(pOutLocal);

581

591  if (((uint32_t) (heightIn == 1) || (uint32_t) (heightIn == 2)) &&
592  ((uint32_t) (strideOut / DSPLIB_sizeof(dataType)) == (uint32_t) heightIn)) {

594  if (heightIn == 1) {

603  interleaveOverheadCycles = 25;

606  interleaveTeardownCycles = 1;
607  }

623  }
