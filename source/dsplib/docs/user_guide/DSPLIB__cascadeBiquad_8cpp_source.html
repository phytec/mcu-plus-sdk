311  computeCycles = (startupCyclesL2 + computeCyclesL1 + teardownCyclesL2) * numChannelGroups;
312 
313  // Entry & exit cycles

315  exitCycles = 8;
316  computeCycles += entryCycles + exitCycles;
317 
318  break;
319 
320  default:

323  }

335 DSPLIB_STATUS DSPLIB_cascadeBiquad_init_checkParams(DSPLIB_kernelHandle handle,
336  const DSPLIB_bufParams2D_t *bufParamsIn,
337  const DSPLIB_bufParams2D_t *bufParamsFilterCoeff,
338  const DSPLIB_bufParams2D_t *bufParamsFilterVar,
339  const DSPLIB_bufParams2D_t *bufParamsOut,
340  const DSPLIB_cascadeBiquad_InitArgs *pKerInitArgs)
341 {
342  DSPLIB_STATUS status = DSPLIB_SUCCESS;
343 
344  if (handle == NULL) {
345  status = DSPLIB_ERR_NULL_POINTER;
346  }
347 
348  if (status == DSPLIB_SUCCESS) {
349  if (bufParamsIn->data_type != DSPLIB_FLOAT32) {
350  status = DSPLIB_ERR_INVALID_TYPE;
351  }
352  else if (bufParamsIn->data_type != bufParamsFilterCoeff->data_type) {
353  status = DSPLIB_ERR_INVALID_TYPE;
354  }
355  else if (bufParamsIn->data_type != bufParamsFilterVar->data_type) {
356  status = DSPLIB_ERR_INVALID_TYPE;
357  }
358  else if (bufParamsIn->data_type != bufParamsOut->data_type) {
359  status = DSPLIB_ERR_INVALID_TYPE;
360  }
361  else {
362  /* Nothing to do here */
363  }
364  }
365 
366  if (status == DSPLIB_SUCCESS) {
367  // Enforce same input and output buffer pitch
368  if (bufParamsIn->stride_y != bufParamsOut->stride_y) {
369  status = DSPLIB_ERR_NOT_IMPLEMENTED;
370  }
371  }
372 
373  if (status == DSPLIB_SUCCESS) {
374  if ((pKerInitArgs->funcStyle != DSPLIB_FUNCTION_NATC) &&
375  (pKerInitArgs->funcStyle != DSPLIB_FUNCTION_OPTIMIZED)) {
376  status = DSPLIB_ERR_INVALID_VALUE;
377  }
378  }
379 
380  if (status == DSPLIB_SUCCESS) {
381  if ((pKerInitArgs->dataSize == 0) ||
382  (pKerInitArgs->numChannels == 0) ||
383  (pKerInitArgs->numStages == 0)) {
384  status = DSPLIB_ERR_INVALID_VALUE;
385  }
386  }
387 
388  if (status == DSPLIB_SUCCESS) {
389  if ((pKerInitArgs->biquadForm != DSPLIB_BIQUAD_TIF) &&
390  (pKerInitArgs->biquadForm != DSPLIB_BIQUAD_DF1) &&
391  (pKerInitArgs->biquadForm != DSPLIB_BIQUAD_DF2) &&
392  (pKerInitArgs->biquadForm != DSPLIB_BIQUAD_TDF2)) {
393  status = DSPLIB_ERR_INVALID_VALUE;
394  }
395  }
396 
397  if (status == DSPLIB_SUCCESS) {
398  if ((pKerInitArgs->enableNChCoeffs != 0) &&
399  (pKerInitArgs->enableNChCoeffs != 1)) {
400  status = DSPLIB_ERR_INVALID_VALUE;
401  }
402  }
403 
404  return status;
405 }

425 DSPLIB_STATUS DSPLIB_cascadeBiquad_init(DSPLIB_kernelHandle handle,
426  DSPLIB_bufParams2D_t *bufParamsIn,
427  DSPLIB_bufParams2D_t *bufParamsFilterCoeff,
428  DSPLIB_bufParams2D_t *bufParamsFilterVar,
429  DSPLIB_bufParams2D_t *bufParamsOut,
430  DSPLIB_cascadeBiquad_InitArgs *pKerInitArgs)
431 {
432  DSPLIB_STATUS status = DSPLIB_SUCCESS;
433  DSPLIB_cascadeBiquad_PrivArgs *pKerPrivArgs = (DSPLIB_cascadeBiquad_PrivArgs *) handle;
434 
435  pKerPrivArgs->initArgs = *pKerInitArgs;
436 
437  int32_t ele_size = 1;
438 
439  if (bufParamsIn->data_type == DSPLIB_FLOAT32) {
440  ele_size = 4;
441  }
442  else {
443  status = DSPLIB_ERR_INVALID_TYPE;
444  }
445 
446  if (status == DSPLIB_SUCCESS) {
447  // Enforce same input and output buffer pitch
448  if (bufParamsIn->stride_y != bufParamsOut->stride_y) {
449  status = DSPLIB_ERR_NOT_IMPLEMENTED;
450  }
451  }
452 
453  if (status == DSPLIB_SUCCESS) {
454  pKerPrivArgs->dataBufferInPitch = bufParamsIn->stride_y / ele_size;
455  pKerPrivArgs->dataBufferOutPitch = bufParamsOut->stride_y / ele_size;
456  pKerPrivArgs->filterVarPitch = bufParamsFilterVar->stride_y / ele_size;
457 
458  if (pKerInitArgs->funcStyle == DSPLIB_FUNCTION_NATC) {
459  // Using Natural-C code
460 
461  status = DSPLIB_cascadeBiquad_init_cn(handle, bufParamsIn,
462  bufParamsFilterCoeff, bufParamsFilterVar, bufParamsOut, pKerInitArgs);
463 
464  if (status == DSPLIB_SUCCESS) {
465  if (pKerInitArgs->biquadForm == DSPLIB_BIQUAD_DF1) {
466  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_df1_exec_cn<float>;
467  }
468  else if (pKerInitArgs->biquadForm == DSPLIB_BIQUAD_DF2) {
469  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_df2_exec_cn<float>;
470  }
471  else if (pKerInitArgs->biquadForm == DSPLIB_BIQUAD_TDF2) {
472  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tdf2_exec_cn<float>;
473  }
474  else if (pKerInitArgs->biquadForm == DSPLIB_BIQUAD_TIF) {
475  // Use DF2 Natural-C code for TI biquad form
476  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_df2_exec_cn<float>;
477  }
478  else {
479  status = DSPLIB_ERR_NOT_IMPLEMENTED;
480  }
481  }
482  }
483  else { // DSPLIB_FUNCTION_OPTIMIZED
484  // Using optimized code
485 
486  switch (pKerInitArgs->biquadForm) {
487  case DSPLIB_BIQUAD_DF1:
488  status = DSPLIB_cascadeBiquad_df1_init_ci<float>(handle, bufParamsIn, bufParamsFilterCoeff,
489  bufParamsFilterVar, bufParamsOut, pKerInitArgs);
490  if (status == DSPLIB_SUCCESS) {
491  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_df1_exec_ci<float>;
492  }
493 
494  break;
495 
496  case DSPLIB_BIQUAD_DF2:
497  status = DSPLIB_cascadeBiquad_df2_init_ci<float>(handle, bufParamsIn, bufParamsFilterCoeff,
498  bufParamsFilterVar, bufParamsOut, pKerInitArgs);
499  if (status == DSPLIB_SUCCESS) {
500  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_df2_exec_ci<float>;
501  }
502 
503  break;
504 
505  case DSPLIB_BIQUAD_TDF2:
506  status = DSPLIB_cascadeBiquad_tdf2_init_ci<float>(handle, bufParamsIn, bufParamsFilterCoeff,
507  bufParamsFilterVar, bufParamsOut, pKerInitArgs);
508  if (status == DSPLIB_SUCCESS) {
509  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tdf2_exec_ci<float>;
510  }
511 
512  break;
513 
514  case DSPLIB_BIQUAD_TIF:
515  if (pKerInitArgs->enableNChCoeffs == 0) {
516  //
517  // Same coefficients for all channels
518  //
519 
520  if (pKerPrivArgs->initArgs.numStages <= DSPLIB_CASCADEBIQUAD_TIF_THR_LOW_STAGE_COUNT) {
521  // Low-stage count implementation
522 
523  status = DSPLIB_cascadeBiquad_tif_lStage_init_ci<float>(handle, bufParamsIn, bufParamsFilterCoeff,
524  bufParamsFilterVar, bufParamsOut, pKerInitArgs);
525 
526  if (status == DSPLIB_SUCCESS) {
527  if (pKerPrivArgs->initArgs.numStages == DSPLIB_CASCADEBIQUAD_NUM_STAGES_1) {
528  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tif_1Stage_exec_ci<float>;
529  }
530  else if (pKerPrivArgs->initArgs.numStages == DSPLIB_CASCADEBIQUAD_NUM_STAGES_2) {
531  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tif_2Stage_exec_ci<float>;
532  }
533  else { // pKerPrivArgs->initArgs.numStages == DSPLIB_CASCADEBIQUAD_NUM_STAGES_3
534  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tif_3Stage_exec_ci<float>;
535  }
536  }
537  }
538  else {
539  // High-stage count implementation
540 
541  status = DSPLIB_cascadeBiquad_tif_hStage_init_ci<float>(handle, bufParamsIn, bufParamsFilterCoeff,
542  bufParamsFilterVar, bufParamsOut, pKerInitArgs);
543 
544  if (status == DSPLIB_SUCCESS) {
545  if (pKerPrivArgs->initArgs.numStages == DSPLIB_CASCADEBIQUAD_NUM_STAGES_4) {
546  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tif_4Stage_exec_ci<float>;
547  }
548  else if (pKerPrivArgs->initArgs.numStages == DSPLIB_CASCADEBIQUAD_NUM_STAGES_5) {
549  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tif_5Stage_exec_ci<float>;
550  }
551  else if (pKerPrivArgs->initArgs.numStages == DSPLIB_CASCADEBIQUAD_NUM_STAGES_6) {
552  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tif_6Stage_exec_ci<float>;
553  }
554  else if (pKerPrivArgs->initArgs.numStages == DSPLIB_CASCADEBIQUAD_NUM_STAGES_8) {
555  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tif_8Stage_exec_ci<float>;
556  }
557  else {
558  // 7 stages is handled with M-stage implementation
559  // More than 8 stages is handled with M-stage implementation
560  pKerPrivArgs->execute = DSPLIB_cascadeBiquad_tif_MStage_exec_ci<float>;
561  }
562  }
563  }
564  }
565  else { // (pKerInitArgs->enableNChCoeffs == 1)
566  //
567  // Different coefficients for each channel
568  //
569 
570  if (pKerPrivArgs->initArgs.numStages <= DSPLIB_CASCADEBIQUAD_TIF_THR_LOW_STAGE_COUNT) {
