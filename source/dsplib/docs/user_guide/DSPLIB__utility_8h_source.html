71 // Define a 128-bit integer class to allow natural-c implementations of DSPLIB
72 // 32-bit input/output functions to be templated. The class is implemented in
73 // a header file for easy sharing. All member functions, including constructors
74 // are declared inline for two reasons: (1) performance and (2) necessary for
75 // implementing the class in a multiple-inclusion header file.
76 
77 class DSPLIB_int128_t {
78  public:
79  int64_t hi;
80  int64_t lo;
81  DSPLIB_int128_t(int64_t h,
82  int64_t l); // constructor for both high and low specified
83  DSPLIB_int128_t(int64_t l); // constructor for just low specified (sign extends to high)
84  DSPLIB_int128_t(); // constructor for neither field specified
85  DSPLIB_int128_t operator+(const DSPLIB_int128_t &) const; // operator +
86  DSPLIB_int128_t operator>>(const int8_t &) const; // operator >>
87 };

112 static inline void DSPLIB_UTIL_shiftRight128(uint64_t *rh, // result
113  uint64_t *rl,
114  uint64_t ah, // operand
115  uint64_t al,
116  int32_t sh, // shift amount
117  int32_t s) // signed
118 {
119  uint64_t h;
120  uint64_t l;
121  int32_t i;
122 
123  h = ah;
124  l = al;
125  for (i = 0; i < sh; i++) {
126  l = (uint64_t) __shift_right(l, (uint32_t) 1) | (uint64_t) __shift_left(h, (uint32_t) 63);
127  h = (uint64_t) __shift_right(h, (uint32_t) 1) | ((s != 0) ? (h & 0x8000000000000000ULL) : 0ULL);
128  }
129 
130  *rh = h;
131  *rl = l;
132 }

134 static inline void DSPLIB_UTIL_Add128(uint64_t *rh, // result
135  uint64_t *rl,
136  uint64_t ah, // left operand
137  uint64_t al,
138  uint64_t bh, // right operand
139  uint64_t bl)
140 {
141  // break up the operands into 4 32b chunks packed into 64b uints
142  uint64_t all;
143  uint64_t alh;
144  uint64_t ahl;
145  uint64_t ahh;
146  uint64_t bll;
147  uint64_t blh;
148  uint64_t bhl;
149  uint64_t bhh;
150  uint64_t s0;
151  uint64_t s1;
152  uint64_t s2;
153  uint64_t s3;
154  uint64_t sh;
155  uint64_t sl;
156 
157  all = (uint64_t) __shift_right(al, (uint32_t) 0) & 0x0ffffffffULL;
158  alh = (uint64_t) __shift_right(al, (uint32_t) 32) & 0x0ffffffffULL;
159  ahl = (uint64_t) __shift_right(ah, (uint32_t) 0) & 0x0ffffffffULL;
160  ahh = (uint64_t) __shift_right(ah, (uint32_t) 32) & 0x0ffffffffULL;
161 
162  bll = (uint64_t) __shift_right(bl, (uint32_t) 0) & 0x0ffffffffULL;
163  blh = (uint64_t) __shift_right(bl, (uint32_t) 32) & 0x0ffffffffULL;
164  bhl = (uint64_t) __shift_right(bh, (uint32_t) 0) & 0x0ffffffffULL;
165  bhh = (uint64_t) __shift_right(bh, (uint32_t) 32) & 0x0ffffffffULL;
166 
167  // the adds
168  s0 = all + bll;
169  s1 = alh + blh + __shift_right(s0, (uint32_t) 32);
170  s2 = ahl + bhl + __shift_right(s1, (uint32_t) 32);
171  s3 = ahh + bhh + __shift_right(s2, (uint32_t) 32);
172 
173  // pack the results
174  sl = (uint64_t) __shift_left(s1, (uint32_t) 32) | (s0 & 0x0ffffffffULL);
175  sh = (uint64_t) __shift_left(s3, (uint32_t) 32) | (s2 & 0x0ffffffffULL);
176 
177  *rl = sl;
178  *rh = sh;
179 }

200 static inline DSPLIB_int128_t DSPLIB_UTIL_mult(int64_t a, int64_t b)
201 {
202  DSPLIB_int128_t result(0, 0);
203 
204  result.lo = (int64_t) a * (int64_t) b;
205  // sign extend the product
206  result.hi =
207  (((uint64_t) result.lo & 0x8000000000000000ULL) != 0LL) ? (int64_t) 0xffffffffffffffffULL : (int64_t) 0ULL;
208 
209  return result;
210 }
