476 static inline void Vhwa_M2mLdcSl2AllocPrmsInit(Vhwa_M2mLdcSl2AllocPrms *sl2Prms)
477 {
478  uint32_t cnt;
479 
480  if (NULL != sl2Prms)
481  {
482  for (cnt = 0u; cnt < LDC_MAX_OUTPUT; cnt ++)
483  {
484  if (0u == cnt)
485  {
486  sl2Prms->enableOutput[cnt] = (uint32_t)UTRUE;
487  sl2Prms->outCcsf[cnt] = FVID2_CCSF_BITS16_PACKED;
488  }
489  else
490  {
491  sl2Prms->enableOutput[cnt] = (uint32_t)UTRUE;
492  sl2Prms->outCcsf[cnt] = FVID2_CCSF_BITS8_PACKED;
493  }
494  sl2Prms->maxNumBlocks[cnt] = 2U;
495  }
496  sl2Prms->maxBlockWidth = 64U;
497  sl2Prms->maxBlockHeight = 64U;
498  #if defined(VHWA_VPAC_IP_REV_VPAC) || defined(VHWA_VPAC_IP_REV_VPAC3)
499  sl2Prms->enable16KBAlignment = (uint32_t)UTRUE; /* 16KB alignment enabled by default (required for firewall support) */
500  #else
501  sl2Prms->enable16KBAlignment = (uint32_t)UFALSE; /* VPAC3L does not support firewall */
502  #endif
503  }
504 }
