423  /* DOF TWO-REGION FIREWALL LAYOUT (uses DMPAC regions 0 and 1)
424  * DMPAC has 3 total regions: DOF uses regions 0-1, SDE uses region 2
425  *
426  * Region 0 (MAIN/DMPAC_FWL_REGION_DOF_MAIN):
427  * Contents: Current image + FOCO preprocessed current + Predictors + Flow output
428  * Permissions: R5(213), FOCO(211), DOF(210)
429  *
430  * Region 1 (REFERENCE/DMPAC_FWL_REGION_DOF_REFERENCE):
431  * Contents: Reference image + FOCO preprocessed reference
432  * Permissions: R5(213), FOCO(211), DOF_RD(209)
433  *
434  * Memory Layout: [Region0: MAIN][Region1: REFERENCE]
435  * Both regions are 16KB-aligned in address and size (firewall requirement)
436  */
437  uint64_t mainRegionSl2Addr;
440  uint32_t mainRegionSl2Size;
444  uint64_t refRegionSl2Addr;
447  uint32_t refRegionSl2Size;
450 } Vhwa_M2mDofSl2Info;
