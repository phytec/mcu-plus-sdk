540  {
541  for(cnt = 0; cnt < VHWA_M2M_MSC_MAX_INST; cnt++)

543  /* LDRA_JUSTIFY_START
544  <metric start> statement branch <metric end>
545  <justification start>
546  Rationale: The component-level test applications already cover this portion of the code. Since coverage cannot be generated for static inline functions in header files, a justification has been provided accordingly.
547  Effect on this unit: None; this portion of the code is already covered by our code base.
548  <justification end> */
549  for(idx = 0; idx < VHWA_M2M_MSC_MAX_IN_CHANNEL; idx++)
550  {
551  sl2Prms->maxInWidth[cnt][idx] = 1920U;
552  sl2Prms->inCcsf[cnt][idx] = FVID2_CCSF_BITS12_UNPACKED16;
553  sl2Prms->inBuffDepth[cnt][idx] = 6U;
554  }
555  /* LDRA_JUSTIFY_END */
556  }
557 
558  for(cnt = 0; cnt < MSC_MAX_OUTPUT; cnt++)
559  {
560  sl2Prms->maxOutWidth[cnt] = 1920U;
561  sl2Prms->outCcsf[cnt] = FVID2_CCSF_BITS12_UNPACKED16;
562  sl2Prms->outBuffDepth[cnt] = 2U;
563  }
564  #if defined(VHWA_VPAC_IP_REV_VPAC) || defined(VHWA_VPAC_IP_REV_VPAC3)
